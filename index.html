@RestController
@RequestMapping("/api/v1/invoices")
public class InvoiceController {

    @Autowired
    private InvoiceService invoiceService;

    // batch endpoint used across billing cycles
    @GetMapping("/batch/{cycleId}")
    public ResponseEntity<List<Invoice>> getBatch(
            @PathVariable String cycleId) {
        return ResponseEntity.ok(
            invoiceService.findByCycle(cycleId));
    }
}
